- [ ] **Step 1: 写 MCP 脚本**

Create `server/session-query-mcp.ts`(镜像 `browser-use-mcp.ts` 的结构,顶部 `#!/usr/bin/env node` + `import './load-env.js';`,复用其 `JsonRpcRequest`/`ToolDefinition`/`textResponse`/`jsonResponse`/`readString`/`readOptionalString`/`readNumber` helpers —— 直接抄或 import)。核心差异在 tools + callTool:

```ts
// 复用 browser-use-mcp.ts 的:JsonRpcRequest, ToolDefinition, textResponse, jsonResponse,
// readString, readOptionalString, readNumber, 以及 stdio JSON-RPC 主循环(initialize/tools/list/tools/call)
// —— 从 browser-use-mcp.ts 抄过来(保持一致)。

const planrlaApiUrl = (process.env.PLANRLA_API_URL || 'http://127.0.0.1:3001').replace(/\/$/, '');
const planrlaApiToken = process.env.SESSION_QUERY_MCP_TOKEN || '';
const API_TIMEOUT_MS = Number.parseInt(process.env.SESSION_QUERY_API_TIMEOUT_MS || '30000', 10);
const MAX_RETURN_MESSAGES = 20;

const tools: ToolDefinition[] = [
  {
    name: 'query_session',
    description: 'Query messages from a referenced chat session on this host. Use to look up what was discussed / decided in another session (cross-session context). Supports keyword search or range.',
    inputSchema: {
      type: 'object',
      properties: {
        sessionId: { type: 'string', description: 'The session id to query.' },
        query: { type: 'string', description: 'Optional case-insensitive substring to search for in message content. Returns matches + 1 message of context around each.' },
        range: {
          type: 'object',
          description: 'Optional message index range [start, end). Use for "last N": { start: total-N, end: total }.',
          properties: {
            start: { type: 'number' },
            end: { type: 'number' },
          },
        },
      },
      required: ['sessionId'],
    },
  },
];

type SessionMessage = { role: string; content: string; timestamp?: string };

async function fetchSessionMessages(sessionId: string): Promise<{ messages: SessionMessage[]; total: number }> {
  if (!planrlaApiToken) {
    throw new Error('SESSION_QUERY_MCP_TOKEN is not configured.');
  }
  const response = await fetch(`${planrlaApiUrl}/api/providers/sessions/${encodeURIComponent(sessionId)}/messages`, {
    method: 'GET',
    headers: { Authorization: `Bearer ${planrlaApiToken}` },
    signal: AbortSignal.timeout(API_TIMEOUT_MS),
  });
  if (!response.ok) {
    throw new Error(`Failed to fetch session messages (${response.status})`);
  }
  const body = await response.json() as { messages?: SessionMessage[]; data?: { messages?: SessionMessage[] } };
  const messages = body.messages ?? body.data?.messages ?? [];
  return { messages, total: messages.length };
}

function filterByQuery(messages: SessionMessage[], query: string): SessionMessage[] {
  const q = query.toLowerCase();
  const matchIdx = new Set<number>();
  messages.forEach((m, i) => {
    if ((m.content || '').toLowerCase().includes(q)) {
      matchIdx.add(i);
      if (i > 0) matchIdx.add(i - 1); // 前后 1 条上下文
      if (i < messages.length - 1) matchIdx.add(i + 1);
    }
  });
  return [...matchIdx].sort((a, b) => a - b).map((i) => messages[i]);
}

async function callTool(name: string, args: Record<string, unknown>) {
  if (name !== 'query_session') {
    throw new Error(`Unknown tool: ${name}`);
  }
  const sessionId = readString(args.sessionId, 'sessionId');
  const { messages, total } = await fetchSessionMessages(sessionId);

  const query = readOptionalString(args.query);
  let result: SessionMessage[];
  if (query) {
    result = filterByQuery(messages, query);
  } else if (args.range && typeof args.range === 'object') {
    const start = typeof (args.range as any).start === 'number' ? (args.range as any).start : 0;
    const end = typeof (args.range as any).end === 'number' ? (args.range as any).end : total;
    result = messages.slice(Math.max(0, start), Math.min(total, end));
  } else {
    result = messages.slice(Math.max(0, total - MAX_RETURN_MESSAGES)); // 兜底:最近 20 条
  }

  const truncated = result.length > MAX_RETURN_MESSAGES;
  if (truncated) result = result.slice(0, MAX_RETURN_MESSAGES);

  return jsonResponse({ messages: result, total, truncated });
}

// 然后:stdio JSON-RPC 主循环 —— **完全照抄 browser-use-mcp.ts** 的 stdin line reader +
// dispatch(initialize → 返回 capabilities;tools/list → 返回 tools;tools/call → callTool + 返回结果)。
// 读 browser-use-mcp.ts 末尾的主循环,逐字搬过来,把 callBrowserUseApi 换成上面的 callTool 调用。
```

```js
// 一次性注册内部 session-query MCP server 给所有支持 stdio MCP 的 provider。
// 幂等:已存在(name 相同)则 upsert 覆盖,不重复。
const SESSION_QUERY_MCP_NAME = 'planrla-session-query';
const registerInternalMcp = async () => {
  try {
    const scriptPath = path.resolve(__dirname, 'session-query-mcp.ts'); // 或 dist-server 编译后路径
    const { providerMcpService } = await import('./modules/providers/services/mcp.service.js');
    for (const provider of providerRegistry.listProviders()) {
      try {
        await provider.mcp.upsertServer({
          name: SESSION_QUERY_MCP_NAME,
          transport: 'stdio',
          command: 'node',
          args: ['--import', 'tsx', scriptPath],
          env: { PLANRLA_API_URL: `http://127.0.0.1:${PORT}`, SESSION_QUERY_MCP_TOKEN: internalMcpToken },
          scope: 'user',
        });
      } catch (err) {
        console.warn(`[session-query-mcp] skipped for ${provider.id}: ${err?.message || err}`);
      }
    }
    console.log(`[session-query-mcp] registered to all providers`);
  } catch (err) {
    console.error('[session-query-mcp] registration failed:', err);
  }
};
registerInternalMcp();
```

export interface SessionReference {
  /** 客户端生成的唯一 id。 */
  id: string;
  sessionId: string;
  title: string;
  provider: string;
  /** ISO date string(展示用)。 */

}

/** 一行提示(注入到 prompt,告诉 agent 有这个 session 可用 query_session 查)。 */
export function formatReferenceHint(ref: SessionReference): string {
  return `[可查询引用:「${ref.title}」(id=${ref.sessionId}, ${ref.provider})] 用 query_session 工具按需查它的内容。`;
}

/** 多引用拼成提示区(每行一个,空行分隔)。 */
export function formatReferenceHints(refs: SessionReference[]): string {
  return refs.map(formatReferenceHint).join('\n\n');
}

替换 hook 内容(删 updateReferenceMessages/trimToFirstN/totalTokens/buildReference;formattedSection → formattedHints;addReference 参数简化为无 mode/messages):
```ts
import { useCallback, useMemo, useState } from 'react';
import { formatReferenceHints, type SessionReference } from '../utils/sessionReferenceFormat';

export interface UseSessionReferencesResult {
  references: SessionReference[];
  /** 拼好的提示行(发送时 prepend 到 content);空返回 ''。 */
  formattedHints: string;
  addReference: (ref: Omit<SessionReference, 'id'>) => void;
  removeReference: (id: string) => void;
  clearReferences: () => void;
}

const newId = () => {
  if (typeof crypto !== 'undefined' && 'randomUUID' in crypto) return crypto.randomUUID();
  return `ref-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
};

export function useSessionReferences(): UseSessionReferencesResult {
  const [references, setReferences] = useState<SessionReference[]>([]);
  const addReference = useCallback((ref: Omit<SessionReference, 'id'>) => {
    setReferences((prev) => [...prev, { ...ref, id: newId() }]);
  }, []);
  const removeReference = useCallback((id: string) => {
    setReferences((prev) => prev.filter((r) => r.id !== id));
  }, []);
  const clearReferences = useCallback(() => setReferences([]), []);
  const formattedHints = useMemo(() => formatReferenceHints(references), [references]);
  return { references, formattedHints, addReference, removeReference, clearReferences };
}
```
