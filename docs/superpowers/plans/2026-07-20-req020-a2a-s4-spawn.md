```ts
createSession(input: { provider: string; projectPath: string }): string {
  const sessionId = crypto.randomUUID();
  db.prepare('INSERT INTO sessions (session_id, provider, project_path, created_at, updated_at) VALUES (?, ?, ?, CURRENT_TIMESTAMP, CURRENT_TIMESTAMP)').run(sessionId, input.provider, input.projectPath);

}
```
