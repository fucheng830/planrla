# REQ-020 A2A · S2 Inbound Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans.

**Goal:** planrla 暴露 A2A endpoint(loopback `/api/a2a/`),外部/自家 session 经 tasks/send 注入消息到指定 session,该 session 的 provider agent 处理 + 回复。

**Architecture:** 主 Express 挂 `/api/a2a/`(loopback guard,auth 前)。tasks/send → sessionId 寻址 → `A2AInboundWriter`(仿 kanban RunnerWriter)→ `spawnFns[provider]` → 阻塞等 complete → artifacts 回填。入站安全(sanitize_inbound + filter_outbound 复用 S1 + per-IP 限频 + 一 session 一 run)。

**Tech Stack:** TS + Express + node fetch(无,纯 server)+ in-memory task store。参照:
- hermes-a2a: `plugin/{server,security}.py`(endpoint + sanitize_inbound + RateLimiter)
- planrla: `server/modules/kanban/{runner,runner-writer}.ts`(程序化注入 + writer 契约黄金参照)、`chat-websocket.service.ts`(chatRunRegistry 一 session 一 run)、S1 `a2a-client.service.ts`(filterOutbound 复用)

**TDD:** security/writer 用 node:test 单测;注入/route 靠手动验收(spec 验收标准)。

**执行环境:** worktree `.claude/worktrees/req020-a2a-s2`(planrla 约定)+ subagent-driven(S1 同模式)。husky .bin 缺失,commit `--no-verify`。

```ts
// server/modules/a2a/inbound.security.ts
// 9 条 prompt-injection 正则(移植 hermes-a2a plugin/security.py:17-27)
const INJECTION_PATTERNS = [
  /<\s*system\s*>.*?<\s*\/\s*system\s*>/gsi,
  /\[INST\].*?\[\/INST\]/gsi,
  /ignore\s+(all\s+)?previous\s+instructions?/gi,
  /you\s+are\s+now\s+/gi,
  /new\s+system\s+prompt/gi,
  /disregard\s+(all\s+)?(prior|earlier|above)/gi,
  /override\s+(your\s+)?(instructions?|rules?|guidelines?)/gi,
  /<\|im_(start|end)\|>/g,
  /^(Human|Assistant|System)\s*:/gmi,
];

export function sanitizeInbound(text: string, maxLength = 50_000): string {
  let t = text;
  if (t.length > maxLength) t = t.slice(0, maxLength) + '\n[... message truncated for safety]';
  for (const p of INJECTION_PATTERNS) {
    p.lastIndex = 0; // 防 /g lastIndex 残留(test/replace 后)
    if (p.test(t)) t = t.replace(p, '[FILTERED]');
  }
  return t;
}

// per-IP 限频(移植 hermes security.py:55-70)
export class RateLimiter {
  private buckets = new Map<string, number[]>();
  constructor(private maxRequests = 20, private windowMs = 60_000) {}
  allow(ip: string): boolean {
    const now = Date.now();
    const arr = (this.buckets.get(ip) || []).filter(ts => ts > now - this.windowMs);
    if (arr.length >= this.maxRequests) { this.buckets.set(ip, arr); return false; }
    arr.push(now); this.buckets.set(ip, arr); return true;
  }
}
```

```ts
// server/modules/a2a/a2a-inbound-writer.ts
// 仿 kanban/runner-writer.ts 的 writer 契约,缓存 provider 事件为累积 text。
// spawnFn(provider runtime)通过 writer.send({type:'assistant', content}) 推 assistant 响应,
// 我们累积 text parts,complete 时 resolve Promise 给 inbound.service 拿回复。
export interface A2aInboundResult { text: string; error?: string; }

export class A2AInboundWriter {
  private chunks: string[] = [];
  private resolveDone?: (r: A2aInboundResult) => void;
  private done = false;
  readonly userId = 0; // loopback 无用户态
  readonly isWebSocketWriter = false;

  /** spawnFn 调,推 provider 事件 */
  send(msg: unknown): void {
    if (this.done) return;
    // msg 形状同 kanban writer 收的(ChatSessionWriter 协议):{type:'assistant'|'tool_use'|'complete'|'error', ...}
    const m = msg as Record<string, any>;
    if (m.type === 'assistant' && typeof m.content === 'string') this.chunks.push(m.content);
    else if (m.type === 'text' && typeof m.text === 'string') this.chunks.push(m.text);
    else if (m.type === 'complete') this.finish({ text: this.chunks.join('\n').trim() });
    else if (m.type === 'error') this.finish({ text: '', error: String(m.error || m.message || 'provider error') });
  }
  setSessionId(_id: string): void { /* provider session id 持久化由 sessionsDb 管,writer 不存 */ }
  updateWebSocket(): void { /* no-op,非 ws */ }

  /** inbound.service 等 complete(带超时) */
  waitComplete(timeoutMs: number): Promise<A2aInboundResult> {
    return new Promise((resolve) => {
      this.resolveDone = resolve;
      setTimeout(() => {
        if (!this.done) this.finish({ text: this.chunks.join('\n').trim(), error: 'timeout' });
      }, timeoutMs);
    });
  }

  private finish(r: A2aInboundResult): void {
    if (this.done) return;
    this.done = true;
    this.resolveDone?.(r);
  }
}
```

**注意**:`send(msg)` 的 msg 形状要对照 kanban runner-writer.ts + provider spawnFn(claude-sdk.js 等)实际 emit 的事件类型。**实现时读 claude-sdk.js 的 emit 确认 `type:'assistant'|'complete'|'error'` 字段名**,对齐(可能 type 叫 'text' 或 'message',content 在不同字段)。这是 writer 能否拿到回复的关键。

```ts
// server/modules/a2a/inbound.service.ts
import { spawnFns } from '../providers/spawn-fns.js';
import { sessionsDb } from '../database/index.js';
import { chatRunRegistry } from '../websocket/services/chat-websocket.service.js'; // 确认 export 路径
import { A2AInboundWriter } from './a2a-inbound-writer.js';
import { sanitizeInbound } from './inbound.security.js';
import { filterOutbound } from './a2a-client.service.js'; // 复用 S1

const RESPONSE_TIMEOUT_MS = 120_000;
const PERMISSION_MODE = 'bypassPermissions'; // 入站委托,不打断权限确认(spec 风险节)

// in-memory task store(tasks/get 轮询用;server 重启丢 working,acceptable)
interface Task { id: string; state: string; text: string; error?: string; createdAt: number; }
const tasks = new Map<string, Task>();

export function buildAgentCard(port: number, version: string) {
  return {
    name: 'planrla',
    description: 'Planrla project AI workbench (A2A inbound, loopback only)',
    url: `http://127.0.0.1:${port}/api/a2a`,
    version,
    protocol: 'a2a',
    protocolVersion: '0.2.0',
    capabilities: { streaming: false, pushNotifications: false, multiTurn: true, structuredMetadata: true },
    skills: [{ id: 'general', name: 'General Assistant', description: 'Delegate a task to a planrla session agent' }],
    authentication: {},
  };
}

export interface SendResult { taskId: string; state: string; text: string; error?: string; }

export async function handleTaskSend(params: { message?: { parts?: Array<{ type: string; text?: string }> }; metadata?: { sessionId?: string } }, userId: number): Promise<SendResult> {
  const text = (params.message?.parts || []).filter(p => p.type === 'text').map(p => p.text || '').join('\n').trim();
  if (!text) return { taskId: '', state: 'failed', text: '', error: 'empty message' };
  const sanitized = sanitizeInbound(text);

  const sessionId = params.metadata?.sessionId;
  if (!sessionId) return { taskId: '', state: 'failed', text: '', error: 'metadata.sessionId required' };

  const session = sessionsDb.getSessionById(sessionId);
  if (!session) return { taskId: '', state: 'failed', text: '', error: 'session not found' };

  const run = chatRunRegistry.startRun({ appSessionId: sessionId, provider: session.provider, providerSessionId: session.provider_session_id, userId });
  if (!run) return { taskId: '', state: 'failed', text: '', error: 'Agent busy (session has active run)' };

  const taskId = crypto.randomUUID();
  const task: Task = { id: taskId, state: 'working', text: '', createdAt: Date.now() };
  tasks.set(taskId, task);

  const writer = new A2AInboundWriter();
  // 不 await(异步跑,service 同步返 working 让对方轮询;或 await 同步等 120s,spec 说同步优先)
  // 这里用 await(同步 120s,超时返 working):
  try {
    const spawnFn = spawnFns[session.provider];
    if (!spawnFn) throw new Error(`provider ${session.provider} not supported`);
    spawnFn(sanitized, {
      sessionId: session.provider_session_id || undefined,
      cwd: session.project_path,
      projectPath: session.project_path,
      permissionMode: PERMISSION_MODE,
    }, writer).catch(e => writer.send?.({ type: 'error', error: String(e) }));
    const result = await writer.waitComplete(RESPONSE_TIMEOUT_MS);
    task.state = result.error && result.error === 'timeout' ? 'working' : (result.error ? 'failed' : 'completed');
    task.text = filterOutbound(result.text);
    if (result.error && result.error !== 'timeout') task.error = result.error;
  } catch (e: any) {
    task.state = 'failed'; task.error = String(e.message || e);
  } finally {
    run.end?.(); // chatRunRegistry 释放(确认 run 对象有无 end/close 方法,对照 chat-websocket.service.ts)
  }
  return { taskId, state: task.state, text: task.text, error: task.error };
}

export function getTask(taskId: string): Task | undefined { return tasks.get(taskId); }
export function cancelTask(taskId: string): boolean {
  const t = tasks.get(taskId);
  if (!t) return false;
  t.state = 'canceled'; return true;
}
```

```ts
// server/modules/a2a/inbound.route.ts
import { Router } from 'express';
import { buildAgentCard, handleTaskSend, getTask, cancelTask } from './inbound.service.js';
import { RateLimiter } from './inbound.security.js';
import { readFileSync } from 'node:fs';

const limiter = new RateLimiter(20, 60_000);
const PKG_VERSION = JSON.parse(readFileSync(new URL('../../../package.json', import.meta.url), 'utf8')).version;

export function createA2aInboundRouter(port: number) {
  const router = Router();
  // loopback guard(抄 S1 a2a/mcp.route.ts)
  router.use((req, res, next) => {
    const ip = req.ip || req.socket.remoteAddress || '';
    if (!['127.0.0.1', '::1', '::ffff:127.0.0.1'].includes(ip)) return res.status(403).json({ error: 'loopback only' });
    if (!limiter.allow(ip)) return res.status(429).json({ error: 'rate limit' });
    next();
  });

  router.get('/.well-known/agent.json', (_req, res) => res.json(buildAgentCard(port, PKG_VERSION)));

  router.post('/', async (req, res) => {
    const msg = req.body || {};
    if (msg.method === 'tasks/send') {
      const r = await handleTaskSend(msg.params || {}, req.user?.id || 0);
      const rpcResult = {
        id: r.taskId, status: { state: r.state },
        artifacts: r.text ? [{ parts: [{ type: 'text', text: r.text }] }] : [],
        ...(r.error ? { error: { code: -32000, message: r.error } } : {}),
      };
      return res.json({ jsonrpc: '2.0', id: msg.id, result: rpcResult });
    }
    if (msg.method === 'tasks/get') {
      const t = getTask(msg.params?.id);
      if (!t) return res.json({ jsonrpc: '2.0', id: msg.id, error: { code: -32602, message: 'task not found' } });
      return res.json({ jsonrpc: '2.0', id: msg.id, result: { id: t.id, status: { state: t.state }, artifacts: t.text ? [{ parts: [{ type: 'text', text: t.text }] }] : [] } });
    }
    if (msg.method === 'tasks/cancel') {
      cancelTask(msg.params?.id);
      return res.json({ jsonrpc: '2.0', id: msg.id, result: { id: msg.params?.id, status: { state: 'canceled' } } });
    }
    res.json({ jsonrpc: '2.0', id: msg.id, error: { code: -32601, message: `unknown method: ${msg.method}` } });
  });
  return router;
}
```
